cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(xnet_canopen_modsim LANGUAGES C CXX)
    find_package(Threads REQUIRED)
endif()

option(XNET_CANOPEN_BUILD_MODSIM_APP "Build the external motor simulation slave" ON)
if(NOT XNET_CANOPEN_BUILD_MODSIM_APP AND NOT XNET_CANOPEN_BUILD_TESTS)
    return()
endif()
set(MODSIM_STACK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
add_library(xnet_canopen_modsim_model STATIC motor.cpp)
target_compile_features(xnet_canopen_modsim_model PUBLIC cxx_std_17 c_std_99)
target_include_directories(xnet_canopen_modsim_model PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}" "${MODSIM_STACK_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/../.."
    "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
target_link_libraries(xnet_canopen_modsim_model PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(xnet_canopen_modsim_model PRIVATE /W4)
else()
    target_compile_options(xnet_canopen_modsim_model PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(XNET_CANOPEN_BUILD_MODSIM_APP)
    add_executable(xnet_canopen_modsim main.cpp)
    target_link_libraries(xnet_canopen_modsim PRIVATE xnet_canopen_modsim_model ${CMAKE_DL_LIBS})
    if(MSVC)
        target_compile_options(xnet_canopen_modsim PRIVATE /W4)
    else()
        target_compile_options(xnet_canopen_modsim PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
